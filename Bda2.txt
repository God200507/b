#!/usr/bin/env python3 

import sys

current_word = None
current_count = 0

for line in sys.stdin:
    try:
        word, count = line.split('\t', 1)
        count = int(count)
    except Exception as e:
        print(f"{e}\n")
        continue

    if current_word == word:
        current_count += count
    else:
        if current_word:
            print(f"{current_word}\t{current_count}")
        current_count = count
        current_word = word

if current_word is not None:
    print(f"{current_word}\t{current_count}")
